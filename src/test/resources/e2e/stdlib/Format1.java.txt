public class Main {
    public static void main(String[] args) {
        System.out.println(String.format("[%d] [%5d] [%-5d] [%05d] [%+d] [% d] [%,d] [%(d]", 42, 42, 42, 42, 42, 42, 1234567, -5));
        System.out.println(String.format("[%d] [%,d] [%08d] [%(d]", -42, -1234567, -42, -42));
        System.out.println(String.format("%d %d", 123456789012L, Long.MIN_VALUE));
        System.out.println(String.format("[%f] [%.0f] [%.1f] [%.3f] [%10.2f] [%-10.2f] [%010.2f]", 3.14159, 2.5, 0.05, 2.0005, 3.14159, 3.14159, -3.14159));
        System.out.println(String.format("%.2f %.2f %.2f %.2f", 0.125, 1.005, 2.675, 0.005));
        System.out.println(String.format("%.2f %.2f %.0f %.0f %.0f", 99.995, 0.0, 0.5, 1.5, 2.5));
        System.out.println(String.format("%.3f %.1f %f", 1e10, 1.0E-5, 1.0E-7));
        System.out.println(String.format("%,.2f %+.1f %(.2f", 1234567.891, 5.55, -3.14159));
        System.out.println(String.format("%.2f %.2f", Double.NaN, 1.0 / 0.0));
        System.out.println(String.format("%x %X %o %#x %08x", 255, 255, 8, 255, 48879));
        System.out.println(String.format("%x %x", -1, -1L));
        System.out.println(String.format("[%s] [%10s] [%-10s] [%.2s] [%S]", "abc", "abc", "abc", "abc", "abc"));
        System.out.println(String.format("[%c] [%c] [%3c]", 'x', 65, 'y'));
        System.out.println(String.format("[%b] [%B] [%b] [%b]", true, false, null, "x"));
        System.out.println(String.format("%s %s %s %s", 1, 2.5, 'c', true));
        System.out.println(String.format("%s", (Object) null));
        System.out.println(String.format("%1$s %2$s %1$s", "a", "b"));
        System.out.println(String.format("100%% done%n").length());
        System.out.println(String.format("no args"));
        System.out.println(String.format("%5s|%-5s|", "toolongvalue", "x"));
        System.out.printf("%d + %d = %d%n", 1, 2, 3);
        System.out.printf("name=%s score=%.1f%n", "bob", 91.25);
        System.out.format("%s-%s%n", "p", "q");
        String s = String.format("%08.3f|%+08d", 3.14159, 42);
        System.out.println(s);
        System.out.println(String.format("%6.2f%%", 12.3456));
        System.out.println(String.join("|", "a", "b", "c"));
        System.out.println(String.join(",", new String[]{"x", "y"}));
        System.out.println(String.join("-"));
    }
}
