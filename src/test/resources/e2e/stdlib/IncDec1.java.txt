public class Main {
    static int s = 5;
    int f = 7;
    static long total = 100L;

    public static void main(String[] args) {
        int c = 0;
        int x = c++;
        System.out.println(x + " " + c);
        int y = ++c;
        System.out.println(y + " " + c);
        int z = --c;
        System.out.println(z + " " + c);
        int w = c--;
        System.out.println(w + " " + c);
        int[] arr = new int[4];
        int i = 0;
        arr[i++] = 10;
        arr[i++] = 20;
        arr[i] = 30;
        System.out.println(arr[0] + " " + arr[1] + " " + arr[2] + " " + i);
        arr[1]++;
        ++arr[2];
        arr[0]--;
        System.out.println(arr[0] + " " + arr[1] + " " + arr[2]);
        int v = arr[1]++;
        System.out.println(v + " " + arr[1]);
        System.out.println(s++ + " " + s);
        System.out.println(++s + " " + s);
        Main o = new Main();
        int old = o.f--;
        System.out.println(old + " " + o.f + " " + (++o.f) + " " + (o.f++) + " " + o.f);
        long l = total++;
        System.out.println(l + " " + total + " " + (--total));
        byte b = 127;
        b++;
        System.out.println(b);
        byte b2 = b--;
        System.out.println(b2 + " " + b);
        short sh = 32767;
        sh++;
        System.out.println(sh);
        char ch = 'a';
        ch++;
        System.out.println(ch);
        char ch2 = ch++;
        System.out.println(ch2 + " " + ch);
        double d = 1.5;
        double dd = d++;
        System.out.println(dd + " " + d + " " + (--d));
        float fl = 0.5f;
        float fo = fl++;
        System.out.println(fo + " " + fl);
        int k = 3;
        int sum = k++ + k++ + ++k;
        System.out.println(sum + " " + k);
        int n = 0;
        for (int j = 0; j < 5; j++) {
            n += j++;
        }
        System.out.println(n);
        int p = 10;
        while (p-- > 7) {
            System.out.print(p + " ");
        }
        System.out.println(p);
        Integer boxed = 5;
        boxed++;
        Integer got = boxed++;
        System.out.println(got + " " + boxed);
    }
}
