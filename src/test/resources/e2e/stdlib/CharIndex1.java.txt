public class Main {
    public static void main(String[] args) {
        int[] counts = new int[128];
        String s = "hello world";
        int i = 0;
        while (i < s.length()) {
            char c = s.charAt(i);
            counts[c] = counts[c] + 1;
            i = i + 1;
        }
        char l = 'l';
        char o = 'o';
        System.out.println(counts[l]);
        System.out.println(counts[o]);
        counts['z'] = 7;
        System.out.println(counts['z']);
        short sh = 3;
        byte by = 4;
        int[] small = {10, 20, 30, 40, 50};
        System.out.println(small[sh] + small[by]);
        small[sh] = 99;
        System.out.println(small[3]);
    }
}
