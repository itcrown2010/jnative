public class Main {
    static int sf = 5;
    int f = 7;

    public static void main(String[] args) {
        int a = 10;
        a += 5;
        System.out.println(a);
        a -= 3;
        System.out.println(a);
        a *= 4;
        System.out.println(a);
        a /= 5;
        System.out.println(a);
        a %= 7;
        System.out.println(a);
        a |= 8;
        System.out.println(a);
        a &= 12;
        System.out.println(a);
        a ^= 5;
        System.out.println(a);
        a <<= 3;
        System.out.println(a);
        a >>= 1;
        System.out.println(a);
        int neg = -64;
        neg >>>= 28;
        System.out.println(neg);

        // implicit narrowing
        char c = 'a';
        c += 2;
        System.out.println(c);
        short s = 30000;
        s += 5000;
        System.out.println(s);
        byte b = 120;
        b += 10;
        System.out.println(b);
        int i = 7;
        i += 2.9;
        System.out.println(i);
        long l = 1L << 40;
        l += 1;
        System.out.println(l);
        double d = 1.5;
        d *= 3;
        System.out.println(d);

        // string
        String str = "x";
        str += "y";
        str += 1;
        System.out.println(str);

        // array / field targets
        int[] arr = new int[3];
        arr[1] += 4;
        arr[1] *= 3;
        System.out.println(arr[1]);
        sf += 10;
        System.out.println(sf);
        Main m = new Main();
        m.f -= 2;
        System.out.println(m.f);
    }
}
